module CarrierBridge {ℓ : Level} {A B : Type ℓ} (f : A → B) where
  carrier-iso : Iso (C.Carrier f) (P.Carrier.Carrier f)
  carrier-iso = C.Carrier-as-Σ f

  carrier-equivalence : C.Carrier f ≃ P.Carrier.Carrier f
  carrier-equivalence = isoToEquiv carrier-iso

  carrier-path : C.Carrier f ≡ P.Carrier.Carrier f
  carrier-path = ua carrier-equivalence

  to-photon : C.Carrier f → P.Carrier.Carrier f
  to-photon = Iso.fun carrier-iso

  descend-commutes : (a : A) → to-photon (C.descend f a) ≡ P.Carrier.descend f a
  descend-commutes a = refl

  ascend-commutes : (c : C.Carrier f) → P.Carrier.ascend f (to-photon c) ≡ C.ascend f c
  ascend-commutes c = refl

  step-commutes : (Φ : A → A) (c : C.Carrier f)
    → to-photon (C.Φ-carrier f Φ c) ≡ P.Carrier.Φ-carrier f Φ (to-photon c)
  step-commutes Φ c = refl

module OrbitBridge {ℓ : Level} {A : Type ℓ} where
  to-photon : O.Orbit A → P.Orbit.Orbit A
  P.Orbit.here (to-photon x) = O.here x
  P.Orbit.next (to-photon x) = to-photon (O.next x)

  from-photon : P.Orbit.Orbit A → O.Orbit A
  O.here (from-photon x) = P.Orbit.here x
  O.next (from-photon x) = from-photon (P.Orbit.next x)

  from-to : (x : O.Orbit A) → from-photon (to-photon x) ≡ x
  O.here (from-to x i) = O.here x
  O.next (from-to x i) = from-to (O.next x) i

  to-from : (x : P.Orbit.Orbit A) → to-photon (from-photon x) ≡ x
  P.Orbit.here (to-from x i) = P.Orbit.here x
  P.Orbit.next (to-from x i) = to-from (P.Orbit.next x) i

  orbit-iso : Iso (O.Orbit A) (P.Orbit.Orbit A)
  orbit-iso = iso to-photon from-photon to-from from-to

  orbit-equivalence : O.Orbit A ≃ P.Orbit.Orbit A
  orbit-equivalence = isoToEquiv orbit-iso

  unfold-commutes : (Φ : A → A) (a : A)
    → to-photon (O.unfold Φ a) ≡ P.Orbit.unfold Φ a
  P.Orbit.here (unfold-commutes Φ a i) = a
  P.Orbit.next (unfold-commutes Φ a i) = unfold-commutes Φ (Φ a) i

module NucleusBridge {ℓ : Level} {A B : Type ℓ} (f : A → B) (Φ : A → A) where
  -- The whole retained trajectory agrees after changing both representations.
  carrier-orbit-bridge : (a : A)
    → OrbitBridge.to-photon
        (O.mapO (CarrierBridge.to-photon f)
          (O.unfold (C.Φ-carrier f Φ) (C.descend f a)))
      ≡ P.Orbit.unfold (P.Carrier.Φ-carrier f Φ) (P.Carrier.descend f a)
  P.Orbit.here (carrier-orbit-bridge a i) = P.Carrier.descend f a
  P.Orbit.next (carrier-orbit-bridge a i) = carrier-orbit-bridge (Φ a) i

module InteractionBridge {ℓ : Level} {W : Type ℓ} (Q : W → Type ℓ)
  (Out : (w : W) → Q w → W → Type ℓ)
  (E : (w : W) (q : Q w) (w' : W) → Out w q w' → Type ℓ) where

  to-photon : {w : W} → S.ISC Q Out E w → P.Dialogue.ISC Q Out E w
  P.Dialogue.react (to-photon p) q =
    let r = S.react p q in
    fst r , fst (snd r) , fst (snd (snd r)) , to-photon (snd (snd (snd r)))

  from-photon : {w : W} → P.Dialogue.ISC Q Out E w → S.ISC Q Out E w
  S.react (from-photon p) q =
    let r = P.Dialogue.react p q in
    fst r , fst (snd r) , fst (snd (snd r)) , from-photon (snd (snd (snd r)))

  from-to : {w : W} (p : S.ISC Q Out E w) → from-photon (to-photon p) ≡ p
  S.react (from-to p i) q =
    let r = S.react p q in
    fst r , fst (snd r) , fst (snd (snd r)) , from-to (snd (snd (snd r))) i

  to-from : {w : W} (p : P.Dialogue.ISC Q Out E w) → to-photon (from-photon p) ≡ p
  P.Dialogue.react (to-from p i) q =
    let r = P.Dialogue.react p q in
    fst r , fst (snd r) , fst (snd (snd r)) , to-from (snd (snd (snd r))) i

  interaction-iso : (w : W) → Iso (S.ISC Q Out E w) (P.Dialogue.ISC Q Out E w)
  interaction-iso w = iso to-photon from-photon to-from from-to

  interaction-equivalence : (w : W) → S.ISC Q Out E w ≃ P.Dialogue.ISC Q Out E w
  interaction-equivalence w = isoToEquiv (interaction-iso w)
