  ----------------------------------------------------------------
  -- ‡ß ¬ the overlap identity:  z = x + iy,  w = u + iv
  ----------------------------------------------------------------
  overlap-identity : (s x y u v : A)
    ‚í (Œ 2 ¬ s + (- x) + (- u)) ¬ (Œ 2 ¬ s + (- x) + (- u)) + (y + (- v)) ¬ (y + (- v))
      ‚â° ((x + (- u)) ¬ (x + (- u)) + (y + (- v)) ¬ (y + (- v))) + Œ 4 ¬ ((s + (- x)) ¬ (s + (- u)))
  overlap-identity s x y u v = shape s x y u v
    where
      shape : (s x y u v : A)
        ‚í ((1r + (1r + 0r)) ¬ s + (- x) + (- u)) ¬ ((1r + (1r + 0r)) ¬ s + (- x) + (- u)) + (y + (- v)) ¬ (y + (- v))
          ‚â° ((x + (- u)) ¬ (x + (- u)) + (y + (- v)) ¬ (y + (- v))) + (1r + (1r + (1r + (1r + 0r)))) ¬ ((s + (- x)) ¬ (s + (- u)))

  ----------------------------------------------------------------

  -- ‚ñF_z ‚àí F_Œz‚ñ¬≤ = 2 ‚àí 2‚ü®F_z,F_Œz‚ü© = 2 ‚àí 2(s‚àíœ)(s+œ)/s¬≤
  bergman-reflection-difference : (s œ : A)

  bergman-reflection-difference s œ = shape s œ
    where

      shape s œ = solve! R

  -- the three curvature readings agree mode by mode:
  -- 4œ¬≤ = 4 s¬≤ det = 2 s¬≤ ‚ñF_z ‚àí F_Œz‚ñ¬≤
  curvature-readings : (s œ det diff : A)

    ‚í (Œ 4 ¬ (œ ¬ œ) ‚â° Œ 4 ¬ ((s ¬ s) ¬ det)) ó (Œ 4 ¬ (œ ¬ œ) ‚â° Œ 2 ¬ ((s ¬ s) ¬ diff))

    cong (Œ 4 ¬_) (sym hdet) , (shape œ ‚àô cong (Œ 2 ¬_) (sym hdiff))
    where

      shape œ = solve! R

  ----------------------------------------------------------------
  -- ‡© ¬ one mode's holonomy defect is 4 sinh¬≤(tœ)
  ----------------------------------------------------------------
  sinh-defect : (c sh : A) ‚í c ¬ c ‚â° 1r + sh ¬ sh
    ‚í ((c + (- sh)) ¬ (c + (- sh)) + (c + sh) ¬ (c + sh)) + (- Œ 2) ‚â° Œ 4 ¬ (sh ¬ sh)

      shape' : (sh : A) ‚í (1r + (1r + 0r)) ¬ (1r + sh ¬ sh) + (1r + (1r + 0r)) ¬ (sh ¬ sh) + (- (1r + (1r + 0r))) ‚â° (1r + (1r + (1r + (1r + 0r)))) ¬ (sh ¬ sh)

    ‚í let m‚‚ = (x‚ + y‚) ¬ (x‚ + y‚) ; m‚‚ = (x‚ + y‚) ¬ (x‚ + y‚) ; m‚‚ = (x‚ + y‚) ¬ (x‚ + y‚)
