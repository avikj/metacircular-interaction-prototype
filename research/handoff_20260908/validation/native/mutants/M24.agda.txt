-- the DMR fibre at n (the body of DMR.RH)
RHAt : ‚ï ‚í Type
RHAt n =
  let a  = fst (DMR.Hfrac (DMR.Œ¥ n))
      b  = snd (DMR.Hfrac (DMR.Œ¥ n))
      n¬≤ = n ¬ n
      n¬≥ = n¬≤ ¬ n

private

  decb-sound (yes a) _ = a
  decb-sound (no ¬a) e = E.rec (false‚âtrue e)

  decb-complete : {A : Type} (d : Dec A) ‚í A ‚í decb d ‚â° true
  decb-complete (yes _) _ = refl
