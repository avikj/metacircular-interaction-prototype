-- the sum over all pairs (a , c) with a + c = n
D : (n : ‚ï) ‚í (‚ï ‚í ‚ï ‚í ‚ï) ‚í ‚ï
D zero    f = f zero zero
D (suc n) f = f zero (suc n) + D n (Œª a c ‚í f (suc a) c)

-- D respects pointwise equality on the pairs it sums

D-ext zero    h = h zero zero refl
D-ext (suc n) h = cong‚ _+_ (h zero (suc n) refl) (D-ext n (Œª a c p ‚í h (suc a) c (cong suc p)))

pow : ‚ï ‚í ‚ï ‚í ‚ï

pow b (suc n) = b ¬ pow b n

private
  pow-+ : (b m n : ‚ï) ‚í pow b (m + n) ‚â° pow b m ¬ pow b n
  pow-+ b zero    n = sym (+-zero (pow b n))
  pow-+ b (suc m) n = cong (b ¬_) (pow-+ b m n) ‚àô ¬-assoc b (pow b m) (pow b n)

  -- D respects pointwise ‚â on the pairs it sums
  D-mono : (n : ‚ï) {f g : ‚ï ‚í ‚ï ‚í ‚ï} ‚í ((a c : ‚ï) ‚í a + c ‚â° n ‚í f a c ‚â g a c) ‚í D n f ‚â D n g
  D-mono zero    h = h zero zero refl
  D-mono (suc n) h = ‚â-+-‚â (h zero (suc n) refl) (D-mono n (Œª a c p ‚í h (suc a) c (cong suc p)))

  -- constants pull out of D
  D-scale : (n k : ‚ï) (f : ‚ï ‚í ‚ï ‚í ‚ï) ‚í D n (Œª a c ‚í k ¬ f a c) ‚â° k ¬ D n f

  D-scale (suc n) k f = cong (k ¬ f zero (suc n) +_) (D-scale n k (Œª a c ‚í f (suc a) c)) ‚àô ¬-distribÀ° k (f zero (suc n)) (D n (Œª a c ‚í f (suc a) c))

  -- k ‚â suc n splits
  ‚â-split‚≤ : (k n : ‚ï) ‚í k ‚â suc n ‚í (k ‚â n) ‚ä (k ‚â° suc n)
  ‚â-split‚≤ k n (zero  , p) = inr p
  ‚â-split‚≤ k n (suc d , p) = inl (d , injSuc p)

  ‚â-¬-‚â : {m n l k : ‚ï} ‚í m ‚â n ‚í l ‚â k ‚í m ¬ l ‚â n ¬ k
  ‚â-¬-‚â {m} {n} {l} {k} p q = ‚â-trans (‚â-¬k p) (subst2 _‚â_ (¬-comm l n) (¬-comm k n) (‚â-¬k q))

-- the Catalan numbers through a memo table (structural in the table size):
-- catTab n k = Cat k for k ‚â n
catTab : ‚ï ‚í ‚ï ‚í ‚ï
catTab zero    k = 1
catTab (suc n) k = decRec (Œª _ ‚í D n (Œª a c ‚í catTab n a ¬ catTab n c)) (Œª _ ‚í catTab n k) (discrete‚ï k (suc n))

cat : ‚ï ‚í ‚ï
cat k = catTab k k

_ : cat 3 ‚â° 5
_ = refl
_ : cat 5 ‚â° 42

private
  -- the table is stable below its size
  stable : (m k : ‚ï) ‚í k ‚â m ‚í catTab m k ‚â° cat k
  stable zero    k k‚â0 = subst (Œª j ‚í catTab zero j ‚â° catTab j j) (sym (‚â0‚í‚â°0 k‚â0)) refl
  stable (suc n) k k‚âsn = lemma (discrete‚ï k (suc n)) refl
    where
      lemma : (d : Dec (k ‚â° suc n)) ‚í discrete‚ï k (suc n) ‚â° d ‚í catTab (suc n) k ‚â° cat k
      lemma (yes p) _ = cong‚ catTab (sym p) refl
      lemma (no  q) e = cong (decRec _ _) e ‚àô stable n k (drop (‚â-split‚≤ k n k‚âsn))
        where
          drop : (k ‚â n) ‚ä (k ‚â° suc n) ‚í k ‚â n
          drop (inl r) = r
          drop (inr r) = ‚ä-elim (q r)

-- the recurrence

cat-suc : (n : ‚ï) ‚í cat (suc n) ‚â° D n (Œª a c ‚í cat a ¬ cat c)

    cong (decRec _ _) (isPropDec (isSet‚ï (suc n) (suc n)) (discrete‚ï (suc n) (suc n)) (yes refl))
  ‚àô D-ext n (Œª a c p ‚í cong‚ _¬_ (stable n a (c , +-comm c a ‚àô p)) (stable n c (a , p)))

module Majorant (x : ‚ï ‚í ‚ï) (b g : ‚ï)
                (x‚ : x 0 ‚â g)
                (step : (n : ‚ï) ‚í x (suc n) ‚â b ¬ D n (Œª a c ‚í x a ¬ x c))
                where

  bound : ‚ï ‚í ‚ï
  bound n = pow b n ¬ (cat n ¬ pow g (suc n))

  private
    -- the product of two bounds at a + c = n is the common scalar times cat a ¬ cat c
    product-of-bounds : (n a c : ‚ï) ‚í a + c ‚â° n
      ‚í bound a ¬ bound c ‚â° (pow b n ¬ pow g (suc (suc n))) ¬ (cat a ¬ cat c)
    product-of-bounds n a c h =
        shape (pow b a) (cat a) (pow g (suc a)) (pow b c) (cat c) (pow g (suc c))
      ‚àô cong‚ (Œª u v ‚í (u ¬ v) ¬ (cat a ¬ cat c))
              (sym (pow-+ b a c) ‚àô cong (pow b) h)
              (sym (pow-+ g (suc a) (suc c)) ‚àô cong (pow g) (cong suc (+-suc a c) ‚àô cong (suc ‚àò suc) h))
      where
        shape : (Ba Ca Ga Bc Cc Gc : ‚ï) ‚í (Ba ¬ (Ca ¬ Ga)) ¬ (Bc ¬ (Cc ¬ Gc)) ‚â° ((Ba ¬ Bc) ¬ (Ga ¬ Gc)) ¬ (Ca ¬ Cc)
        shape Ba Ca Ga Bc Cc Gc = solve‚ï!

    below : (n : ‚ï) (k : ‚ï) ‚í k ‚â n ‚í x k ‚â bound k
    below zero k k‚â0 = subst (Œª k ‚í x k ‚â bound k) (sym (‚â0‚í‚â°0 k‚â0)) (subst (x 0 ‚â_) (sym base) x‚)
      where

        base = solve‚ï!
    below (suc n) k k‚âsn with ‚â-split‚≤ k n k‚âsn
    ... | inl k‚ân = below n k k‚ân
    ... | inr k‚â°sn = subst (Œª k ‚í x k ‚â bound k) (sym k‚â°sn) top
      where
        top : x (suc n) ‚â bound (suc n)
        top = ‚â-trans (step n)
                (subst (b ¬ D n (Œª a c ‚í x a ¬ x c) ‚â_) closed
                       (‚â-¬-‚â (‚â-refl {b}) (D-mono n (Œª a c p ‚í ‚â-¬-‚â (below n a (‚â-of a c p)) (below n c (‚â-of' a c p))))))
          where
            ‚â-of : (a c : ‚ï) ‚í a + c ‚â° n ‚í a ‚â n
            ‚â-of a c p = c , +-comm c a ‚àô p
            ‚â-of' : (a c : ‚ï) ‚í a + c ‚â° n ‚í c ‚â n
            ‚â-of' a c p = a , p
            closed : b ¬ D n (Œª a c ‚í bound a ¬ bound c) ‚â° bound (suc n)
            closed = cong (b ¬_) (D-ext n (product-of-bounds n) ‚àô D-scale n (pow b n ¬ pow g (suc (suc n))) (Œª a c ‚í cat a ¬ cat c)
                                  ‚àô cong ((pow b n ¬ pow g (suc (suc n))) ¬_) (sym (cat-suc n)))
                   ‚àô shape (pow b n) (pow g (suc (suc n))) (cat (suc n))
              where
                shape : (P G C : ‚ï) ‚í b ¬ ((P ¬ G) ¬ C) ‚â° (b ¬ P) ¬ (C ¬ G)
                shape P G C = solve‚ï!

  -- THE MAJORANT:  x‚ô ‚â b‚ø ¬ Cat‚ô ¬ g‚ø‚∫¬
  catalan-majorant : (n : ‚ï) ‚í x n ‚â pow b n ¬ (cat n ¬ pow g n)
  catalan-majorant n = below n n ‚â-refl
